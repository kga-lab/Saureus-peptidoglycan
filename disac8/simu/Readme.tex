%%%%%%%%% get initial configuration for glycan  and keep glycan at the center of the box
1) cp glycan.gro from iniconfig folder.
2) gmx editconf -f glycan.gro -box 11 11 11 -center 5.5 5.5 5.5 -o glycan_fix.gro -resnr 1
%%%%%%%%%%%solvate
3) gmx solvate -cp glycan_fix.gro -cs spc216.gro -p topol.top -o system_sol.gro 
 change in topol.top the name of SOL => TIP3, and in system_sol.gro HW1=> H1, HW2=>H2, OW=>OH2
%%%%%%%%%%%%% since net charge = 0, no ions in the system %%%%%%%%%%%%%
%%%%%%%%%%%%%%%make index %%%%%%%%%
4) gmx make_ndx -f system_sol.gro -o index.ndx 
  name the group for glycan atoms : glycan

%%%%%%%%%%%%%%%Energy Minimization%%%%%%%%%
5) gmx grompp -f step1_mini.mdp -c system_sol.gro -p topol.top -n index.ndx -o system_mini.tpr -r system_sol.gro 

%%%%%%%%%%%%%%Equilibration %%%%%%%%%%%%%
6) gmx grompp -f step2_eq1.mdp -c system_mini.gro -p topol.top -n index.ndx -o system_eq1.tpr -r system_mini.gro 
    run this on tue

7) gmx grompp -f step2_eq2.mdp -c system_eq1.gro -p topol.top -t system_eq1.cpt  -n index.ndx -o system_eq2.tpr -r system_eq1.gro 
    run this on tue

8) gmx grompp -f step3_eq3.mdp -c system_eq2.gro -p topol.top -t system_eq2.cpt -n index.ndx -o system_eq3.tpr  (this is 5 ns npt without restraints)

9) gmx grompp -f step4_prod.mdp -c system_eq3.gro -p topol.top -t system_eq3.cpt -n index.ndx -o system_prod.tpr 
(400 ns)
